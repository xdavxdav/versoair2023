            } else {
                // Scrolling up, show navbar

            }
            lastScrollTop = currentScroll <= 0 ? 0 : currentScroll; // Avoid negative scrolling values
        });

      // Get the scroll position

      const docHeight = document.documentElement.scrollHeight - window.innerHeight;
      const scrollPercentage = scrollTop / docHeight;

      // Calculate how far the text should move based on scroll position
      const windowWidth = window.innerWidth;
      const maxTranslateX = windowWidth; // Move the text from full right (100vw) to full left (-100vw)
