## Quick Start

### Using Custom Hooks
```tsx
// Import and use in any component
import { useDatabase } from "@/hooks/useDatabase";
import { useTableOperations } from "@/hooks/useTableOperations";

const { health, tables, executeQuery } = useDatabase();
const { tableData, addRecord, deleteRecord } = useTableOperations();
```

### Using Accessibility Utilities
```tsx
import { useEscapeKey, generateAriaLabel } from "@/utils/a11y";

// Close modal with Escape key
useEscapeKey(() => setModalOpen(false), isModalOpen);

// Generate descriptive aria-label
<Button aria-label={generateAriaLabel("Delete record")}>Delete</Button>
```

### Using Constants
```tsx
import { DB_CONFIG, PAGINATION_CONFIG, KEYBOARD_SHORTCUTS } from "@/constants/database";

const apiUrl = DB_CONFIG.API_BASE_URL;
const pageSize = PAGINATION_CONFIG.DEFAULT_PAGE_SIZE;
```
